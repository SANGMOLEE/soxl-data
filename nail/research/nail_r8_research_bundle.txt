import pandas as pd, numpy as np
df = pd.read_pickle('nail.pkl')
def rd(p, col='value'):
    x = pd.read_csv(p, parse_dates=['date']).set_index('date')[col]
    return pd.to_numeric(x, errors='coerce').dropna()
vix = rd('gh/VIXCLS.csv'); hy = rd('gh/BAMLH0A0HYM2.csv'); fg = rd('gh/fear_greed.csv')
spy = pd.read_csv('gh/SPY.csv', parse_dates=['date']).set_index('date')['adj_close']
# align to NAIL trading days (use last available value, as-of)
idx = df.index
def asof(s, lag_days=0):
    s = s.sort_index()
    out = s.reindex(s.index.union(idx)).ffill().reindex(idx)
    return out
df['vix'] = asof(vix); df['hy']=asof(hy); df['fg']=asof(fg); df['spy']=asof(spy)
c = df.close
df['dd252'] = c/c.rolling(252, min_periods=60).max()-1
df['dd756'] = c/c.rolling(756, min_periods=60).max()-1
df['ma200'] = c/c.rolling(200, min_periods=100).mean()-1
df['ma50'] = c/c.rolling(50, min_periods=30).mean()-1
def rsi(s, n=14):
    d = s.diff(); up = d.clip(lower=0); dn = -d.clip(upper=0)
    au = up.ewm(alpha=1/n, adjust=False).mean(); ad = dn.ewm(alpha=1/n, adjust=False).mean()
    return 100 - 100/(1+au/ad)
df['rsi14'] = rsi(c)
df['vol20'] = np.log(c).diff().rolling(20).std()*np.sqrt(252)
df['vix60'] = df.vix.rolling(60, min_periods=20).max()
df.to_pickle('nail_feat.pkl')

import pandas as pd, numpy as np, itertools, time, sys
df = pd.read_pickle('nail_feat2.pkl')
idx = df.index; pos = {d:i for i,d in enumerate(idx)}
C = df.close.values; O = df.open.values; L = df.low.values; DD = df.dd252.values; n=len(idx)
FR=0.00075
EV7 = [pos[pd.Timestamp(x)] for x in ['2016-02-11','2016-11-02','2018-12-24','2020-03-23','2022-10-20','2023-10-25','2025-04-08']]
DEEP = [0,2,3,4,6]; CORE=[2,3,4,5,6]
ST = pos[idx[idx>=pd.Timestamp('2016-01-01')][0]]
def bands(x, cuts, pts):
    out = np.zeros(len(x))
    for c,p in zip(cuts,pts): out = np.where((x<=c)&(p>out), p, out)
    return out
def sim(s, ths, step=0.0, reset_dd=-0.30):
    ntr=len(ths); used=0; last=None; camp=0; F=[]; resets={}
    for i in range(ST, n-1):
        if used>0 and DD[i]>reset_dd:
            resets[camp]=i; used=0; last=None; camp+=1
        while used<ntr:
            if s[i] < ths[used]: break
            if step>0 and last is not None and C[i] > last*(1-step): break
            px=O[i+1]*(1+FR); F.append((camp,i+1,px)); last=px; used+=1
            if step>0: break
    return F, resets, used
def evaluate(F, resets, ntr=3):
    # campaigns
    camps={}
    for cid,fi,px in F: camps.setdefault(cid,[]).append((fi,px))
    res=[]; matched=set()
    for k,ie in enumerate(EV7):
        best=None
        for cid,fl in camps.items():
            i0=fl[0][0]
            alive = (cid not in resets) or (resets[cid]>ie)
            if i0<=ie+21 and i0>=ie-252 and alive:
                if best is None or i0>camps[best][0][0]: best=cid
        tc=C[ie]; i1=min(n-1,ie+126); perfect=C[i1]/tc
        if best is None:
            res.append((k,np.nan,np.nan,0.0,0.0,1.0,perfect,None)); continue
        matched.add(best); fl=[x for x in camps[best] if x[0]<=ie+21]
        pxs=np.array([p for _,p in fl]); dep=len(fl)/ntr
        first=pxs[0]/tc-1; avg=(len(pxs)/np.sum(1/pxs))/tc-1  # equal-dollar tranches -> harmonic mean cost
        i0=camps[best][0][0]; sh=np.zeros(i1-i0+1); cash=np.ones(i1-i0+1)
        for fi,px in camps[best]:
            j=fi-i0
            if 0<=j<=i1-i0: sh[j:]+= (1/ntr)/px; cash[j:]-=1/ntr
        val=cash+sh*L[i0:i1+1]; vend=cash[-1]+sh[-1]*C[i1]
        res.append((k,first,avg,dep,val.min()-1,vend,perfect,camps[best]))
    un=[]
    for cid,fl in camps.items():
        if cid in matched: continue
        i0=fl[0][0]; j=min(n-1,i0+126); avgpx=len(fl)/np.sum([1/p for _,p in fl])
        un.append((cid,i0,len(fl),avgpx,C[j]/avgpx-1,L[i0:j+1].min()/fl[0][1]-1,i0+126<n))
    return res, un
def summarize(res, un):
    deep=[res[k] for k in DEEP]
    miss=sum(1 for r in deep if np.isnan(r[2]))
    cap=np.array([ (r[5]-1)/(r[6]-1) for r in deep])
    core=[res[k] for k in CORE]
    near10=sum(1 for r in core if not np.isnan(r[1]) and r[1]<=0.10)
    nfalse=sum(1 for u in un if u[6] and u[4]<=-0.30)
    return dict(miss=miss, cap_med=float(np.median(cap)), cap_min=float(cap.min()), cap=np.round(cap,2).tolist(),
                avg_med=float(np.nanmedian([r[2] for r in deep])), avg_max=float(np.nanmax([r[2] for r in deep])),
                first_med=float(np.nanmedian([r[1] for r in deep])),
                dep=float(np.mean([r[3] for r in deep])), bmae=float(min(r[4] for r in deep)), near10=near10,
                nfalse=nfalse, nunm=len(un))

def simw(s, ths, step, reset_dd=-0.30, dd_reset=None):
    DDr = DD if dd_reset is None else dd_reset
    ntr=len(ths); used=0; last=None; camp=0; F=[]; resets={}
    for i in range(ST, n-1):
        if used>0 and DDr[i]>reset_dd: resets[camp]=i; used=0; last=None; camp+=1
        while used<ntr:
            if s[i] < ths[used]: break
            if step>0 and last is not None and C[i] > last*(1-step): break
            px=O[i+1]*(1+FR); F.append((camp,i+1,px,used)); last=px; used+=1
            if step>0: break
    return F, resets, used
def evalw(F, resets, w):
    camps={}
    for cid,fi,px,k in F: camps.setdefault(cid,[]).append((fi,px,k))
    out=[]; matched=set()
    for kk,ie in enumerate(EV7):
        best=None
        for cid,fl in camps.items():
            i0=fl[0][0]; alive=(cid not in resets) or (resets[cid]>ie)
            if i0<=ie+21 and i0>=ie-252 and alive:
                if best is None or i0>camps[best][0][0]: best=cid
        tc=C[ie]; i1=min(n-1,ie+126); perfect=C[i1]/tc
        if best is None: out.append((kk,np.nan,np.nan,0,0,1,perfect)); continue
        matched.add(best); fl=camps[best]
        i0=fl[0][0]; sh=np.zeros(i1-i0+1); cash=np.ones(i1-i0+1)
        for fi,px,k in fl:
            j=fi-i0
            if 0<=j<=i1-i0: sh[j:]+=w[k]/px; cash[j:]-=w[k]
        fl2=[x for x in fl if x[0]<=ie+21]
        dep=sum(w[k] for _,_,k in fl2); cost=dep/sum(w[k]/px for _,px,k in fl2)
        val=cash+sh*L[i0:i1+1]; vend=cash[-1]+sh[-1]*C[i1]
        out.append((kk, fl2[0][1]/tc-1, cost/tc-1, dep, val.min()-1, vend, perfect))
    un=[cid for cid in camps if cid not in matched]
    deep=[out[k] for k in DEEP]
    cap=np.array([(r[5]-1)/(r[6]-1) for r in deep])
    return dict(miss=sum(np.isnan(r[2]) for r in deep), cap_med=np.median(cap), cap=np.round(cap,2).tolist(),
                avg=np.round([r[2] for r in deep],2).tolist(), dep=np.round([r[3] for r in deep],2).tolist(),
                bmae=min(r[4] for r in deep), nunm=len(un))

def r73_campaigns(sig_dates):
    """Approximate R7.3 campaigns: 1st at given signal date (fill next open); 2nd when T>=10, 3rd when T>=18 (signal->next open),
