# 최저점 매수 최적화용 공통 모듈: 자료, 시뮬레이터, '사이클 최저점 대비 평균 매수 위치' 평가
import pandas as pd, numpy as np
f = pd.read_pickle('nail_feat2.pkl')
v11 = pd.read_csv('r8/v11_chart/r8_daily.csv', parse_dates=['date']).set_index('date')
df = f[['open','high','low','close','A','ma50','vol20','dd252']].copy()
df['S'] = v11['S'].reindex(df.index).fillna(0)
C = df.close.values; O = df.open.values; L = df.low.values; idx = df.index; n = len(df)
DD = (df.close / df.close.rolling(252, min_periods=60).max() - 1).values
def bands(x, cuts=(-0.5,-0.6,-0.7,-0.8,-0.9), pts=(10,20,30,40,50)):
    x = np.asarray(x, float); out = np.zeros(len(x))
    for c, p in zip(cuts, pts): out = np.where((x <= c) & (p > out), p, out)
    return out
N = bands(DD); A = df.A.values.astype(float); S = df.S.values.astype(float)
COND = N + A + S                                   # R8 조건점수 (표시점수 = 100 - COND)
ROC10 = df.close.pct_change(10).values; ROC20 = df.close.pct_change(20).values
TR = pd.to_datetime(['2016-02-11','2016-11-02','2018-12-24','2020-03-23','2022-10-20','2023-10-25','2025-04-08'])
TI = np.array([idx.get_loc(t) for t in TR]); DEEP = [0,2,3,4,6]; H = 21
ST = int(np.searchsorted(idx, pd.Timestamp('2016-01-01')))
def sim(cond, th=(42,52,62), step=0.15, w=(1/3,1/3,1/3), gap=0, knife=None, confirm=False, reset=-0.30):
    used = 0; last = None; lastf = -999; camp = 0; F = []; pend = False; resets = []
    for i in range(ST, n):
        if pend:
            if np.isfinite(O[i]):
                px = O[i] * 1.00075; F.append((camp, i, px, used, w[used])); last = px; used += 1; lastf = i
            pend = False
        if used > 0 and DD[i] > reset: resets.append((camp, i)); used = 0; last = None; camp += 1
        if used < 3 and i + 1 < n and cond[i] >= th[used]:
            ok = used == 0 or C[i] <= last * (1 - step)
            if ok and gap and used > 0 and i - lastf < gap: ok = False
            if ok and knife is not None and knife[i]: ok = False
            if ok and confirm and used == 2 and not (C[i] > C[i-5:i].max()): ok = False
            if ok: pend = True
    return F, dict(resets)
def evaluate(F, resets, h=H):
    """사이클별 평균 매수 위치(저점 종가 대비). 저점+h 거래일까지 못 채운 몫은 그날 종가에 산 것으로 강제 완료."""
    per = []; cur = []
    groups = {k: [] for k in range(len(TI))}
    for (camp, i, px, tr, w) in F:
        k = next((k for k in range(len(TI)) if i <= TI[k] + h), None)
        if k is None: cur.append((camp, i, px, tr, w)); continue
        groups[k].append((camp, i, px, tr, w))
    for k in range(len(TI)):
        g = groups[k]; te = TI[k] + h; low = C[TI[k]]
        W = [x[4] for x in g]; P = [x[2] for x in g]; actual_w = sum(W)
        forced = 0.0
        if not g: forced = 1.0
        else:
            lc = g[-1][0]; rs = resets.get(lc)
            if rs is None or rs > te:
                forced = 1.0 - sum(x[4] for x in F if x[0] == lc)
        if forced > 1e-9: W = W + [forced]; P = P + [C[te]]
        avg = sum(W) / sum(wi / pi for wi, pi in zip(W, P))
        act = (sum(x[4] for x in g) / sum(x[4] / x[2] for x in g)) if g else np.nan
        per.append(dict(k=k, prem=avg / low - 1, act_prem=(act / low - 1) if g else np.nan, actual_w=actual_w, forced=forced,
                        near30=any(x[2] <= low * 1.3 for x in g if x[1] >= TI[k] - 126)))
    return per, cur
def summary(per):
    p = np.array([x['prem'] for x in per]); d = p[DEEP]
    return dict(J7=p.mean(), J5=d.mean(), med7=np.median(p), near30=sum(x['near30'] for x in per), prem=p)

def pts(x, cuts, p, ge=False):
    x = np.asarray(x, float); out = np.zeros(len(x))
    for c, q in zip(cuts, p): out = np.where(((x >= c) if ge else (x <= c)) & (q > out), q, out)
    return np.nan_to_num(out)

def weekly_asof(s, lag=5):
    s = s.dropna().sort_index(); out = []

        if k - lag < 0: out.append(np.nan); continue
        j = s.index.searchsorted(idx[k - lag], side='right') - 1; out.append(s.iloc[j] if j >= 0 else np.nan)
    return np.array(out)
