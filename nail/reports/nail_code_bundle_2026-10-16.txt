# -*- coding: utf-8 -*-
"""
NAIL 연구용 원자료 수집기 (GitHub Actions에서 실행 — 작업 컨테이너에서는 시세 사이트가 막혀 있음)

- 주가: Yahoo(yfinance), 전 기간을 매번 새로 받는다(조정종가가 새 분배금마다 과거까지 다시 계산되므로
  append-only로 쌓으면 총수익 비율 A2가 틀어진다). 결과: nail/data/prices/{이름}.csv
- 거시: FRED fredgraph.csv, 전 기간. 결과: nail/data/macro/{ID}.csv
  HY OAS(BAMLH0A0HYM2)는 FRED가 최근 3년만 보내므로 soxl-data의 보존 이력(data/macro)과 합친다.
- 분배금: nail/data/dividends/{이름}.csv
- 상태: nail/data/fetch_status.json (시리즈별 행수·마지막 날짜·실패)
- 처음 한 번만: soxl-data 공용 수집기용 시드 파일(data/prices/NAIL.csv, ITB.csv, data/macro/DGS10.csv 등)을
  파일이 없을 때만 만든다. 이미 있으면 건드리지 않는다(공용 수집기의 '기존 행 불변' 원칙).
"""
import os, sys, json, time
from datetime import datetime, timezone, timedelta
import pandas as pd

ROOT = os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
OUT = os.path.join(ROOT, 'nail', 'data')
KST = timezone(timedelta(hours=9)); NOW = datetime.now(KST)

# 이름: Yahoo 심볼
PRICES = {

FRED = [

SEED_PRICES = ('NAIL', 'ITB')                     # soxl-data 공용 수집기 시드(없을 때만)
SEED_FRED = ('DGS10', 'DGS2', 'MORTGAGE30US')
status = dict(run_at_kst=NOW.strftime('%Y-%m-%d %H:%M:%S'), prices={}, macro={}, failed=[], notes=[])


def fetch_yahoo(sym):
    import yfinance as yf
    last = None
    for k in range(3):
        try:
            t = yf.Ticker(sym)
            raw = t.history(period='max', interval='1d', auto_adjust=False, actions=True)
            if raw is None or raw.empty: raise ValueError('빈 응답')
            idx = pd.to_datetime(raw.index)
            idx = idx.tz_localize(None) if idx.tz is not None else idx
            d = pd.DataFrame({'date': idx.normalize(), 'open': raw['Open'].values, 'high': raw['High'].values,
                              'low': raw['Low'].values, 'close': raw['Close'].values,
                              'adj_close': (raw['Adj Close'] if 'Adj Close' in raw else raw['Close']).values,
                              'volume': raw['Volume'].values if 'Volume' in raw else 0})
            d = d.dropna(subset=['close']).drop_duplicates('date', keep='last').sort_values('date')
            # 장중 실행이면 오늘 행은 미확정 값이므로 버린다(미국 동부 16:30 이전)
            et = datetime.now(timezone.utc).astimezone(__import__('zoneinfo').ZoneInfo('America/New_York'))
            if len(d) and d.date.iloc[-1].date() == et.date() and (et.hour, et.minute) < (16, 30):
                status['notes'].append(f'{sym}: 장중 미확정 행 {et.date()} 제외'); d = d.iloc[:-1]
            div = raw['Dividends'] if 'Dividends' in raw else pd.Series(dtype=float)
            div = div[div > 0]
            dv = pd.DataFrame({'date': pd.to_datetime(div.index).tz_localize(None).normalize() if len(div) else [],
                               'dividend': div.values})
            spl = raw['Stock Splits'] if 'Stock Splits' in raw else pd.Series(dtype=float)
            spl = spl[spl > 0]
            return d.reset_index(drop=True), dv, [(str(pd.Timestamp(i).date()), float(v)) for i, v in spl.items()]
        except Exception as e:
            last = e; time.sleep(3 + 3 * k)
    raise last


def fetch_fred(sid):
    url = f'https://fred.stlouisfed.org/graph/fredgraph.csv?id={sid}&cosd=1776-07-04&coed={NOW:%Y-%m-%d}'
    last = None
    for k in range(3):
        try:
            raw = pd.read_csv(url)
            raw.columns = ['date', 'value']
            raw['date'] = pd.to_datetime(raw['date']); raw['value'] = pd.to_numeric(raw['value'], errors='coerce')
            raw = raw.dropna().sort_values('date').reset_index(drop=True)
            if raw.empty: raise ValueError('전량 결측')
            return raw
        except Exception as e:
            last = e; time.sleep(2 + 2 * k)
    raise last


def main():
    for sub in ('prices', 'macro', 'dividends'):
        os.makedirs(os.path.join(OUT, sub), exist_ok=True)
    for name, sym in PRICES.items():
        try:
            d, dv, spl = fetch_yahoo(sym)
            d.to_csv(os.path.join(OUT, 'prices', f'{name}.csv'), index=False, float_format='%.6f')
            if len(dv): dv.to_csv(os.path.join(OUT, 'dividends', f'{name}.csv'), index=False, float_format='%.6f')
            status['prices'][name] = dict(symbol=sym, rows=len(d), first=str(d.date.min().date()), last=str(d.date.max().date()),
                                         last_close=round(float(d.close.iloc[-1]), 4), splits=spl)
            print(f'[OK] {name:5s} {len(d):6d}행 {d.date.min().date()} ~ {d.date.max().date()} 종가 {d.close.iloc[-1]:.2f}')
            if name in SEED_PRICES:
                seed = os.path.join(ROOT, 'data', 'prices', f'{name}.csv')
                if not os.path.exists(seed):
                    s = d.copy(); s['source'] = 'yahoo'
                    s.to_csv(seed, index=False); status['notes'].append(f'시드 생성 {seed}')
        except Exception as e:
            status['failed'].append(f'{name}: {type(e).__name__} {str(e)[:80]}'); print(f'[FAIL] {name}: {e}')
        time.sleep(1)
    for sid in FRED:
        try:
            d = fetch_fred(sid)
            if sid == 'BAMLH0A0HYM2':        # FRED 절단 → soxl-data 보존 이력과 합침
                keep = os.path.join(ROOT, 'data', 'macro', 'BAMLH0A0HYM2.csv')
                if os.path.exists(keep):
                    old = pd.read_csv(keep, usecols=[0, 1]); old.columns = ['date', 'value']
                    old['date'] = pd.to_datetime(old['date']); old['value'] = pd.to_numeric(old['value'], errors='coerce')
                    old = old.dropna()
                    d = pd.concat([old[old.date < d.date.min()], d]).drop_duplicates('date', keep='last').sort_values('date')
                    status['notes'].append(f'HY: 보존 이력 {len(old)}행 + FRED {len(d)}행 병합')
            d.to_csv(os.path.join(OUT, 'macro', f'{sid}.csv'), index=False)
            status['macro'][sid] = dict(rows=len(d), first=str(d.date.min().date()), last=str(d.date.max().date()),
                                        last_value=float(d.value.iloc[-1]))
            print(f'[OK] {sid:14s} {len(d):6d}행 ~ {d.date.max().date()} {d.value.iloc[-1]}')
            if sid in SEED_FRED:
                seed = os.path.join(ROOT, 'data', 'macro', f'{sid}.csv')
                if not os.path.exists(seed):
                    s = d.copy(); s['source'] = 'fred'; s.to_csv(seed, index=False); status['notes'].append(f'시드 생성 {seed}')
        except Exception as e:
            status['failed'].append(f'{sid}: {type(e).__name__} {str(e)[:80]}'); print(f'[FAIL] {sid}: {e}')
        time.sleep(0.6)
    json.dump(status, open(os.path.join(OUT, 'fetch_status.json'), 'w', encoding='utf-8'), ensure_ascii=False, indent=1)
    print('실패:', status['failed'])
    core = [k for k in ('NAIL', 'ITB', 'SPY') if k not in status['prices']] + \
           [k for k in ('VIXCLS', 'BAMLH0A0HYM2') if k not in status['macro']]
    if core:
        print('::error::핵심 시리즈 실패', core); sys.exit(1)


if __name__ == '__main__':
    main()

"""

"""
import os, sys, json, subprocess

N = os.path.join(ROOT, 'nail'); D = os.path.join(N, 'data'); O = os.path.join(N, 'output'); E = os.path.join(O, 'engine')
KST = timezone(timedelta(hours=9))


def _md(df):
    cols = list(df.columns)
    out = ['| ' + ' | '.join(cols) + ' |', '|' + '---|' * len(cols)]
    out += ['| ' + ' | '.join(str(v) for v in r) + ' |' for r in df.itertuples(index=False)]
    return '\n'.join(out)


def main():
    os.makedirs(E, exist_ok=True)
    cmd = [sys.executable, os.path.join(N, 'engine', 'nail_r9_engine_v1.2.py'),
           '--nail', f'{D}/prices/NAIL.csv', '--itb', f'{D}/prices/ITB.csv', '--spy', f'{D}/prices/SPY.csv',
           '--vix', f'{D}/macro/VIXCLS.csv', '--hy', f'{D}/macro/BAMLH0A0HYM2.csv', '--out', E]
    try:
        import exchange_calendars  # noqa: F401
    except Exception:
        cmd += ['--calendar-from', f'{D}/prices/SPY.csv']
    r = subprocess.run(cmd, capture_output=True, text=True)
    print(r.stdout); print(r.stderr, file=sys.stderr)
    if r.returncode != 0: sys.exit(r.returncode)
    st = json.load(open(os.path.join(E, 'r8_status.json'), encoding='utf-8'))
    daily = pd.read_csv(os.path.join(E, 'r8_daily.csv'), index_col=0, parse_dates=True)
    last = daily.iloc[-1]; d = daily.index[-1]

               nail_close=round(float(last.close), 4), display=st['display_score'], cond=st['condition_score'],
               N=last.N, A1=last.get('A1'), A2=last.get('A2'), A3=last.get('A3'), A4=last.get('A4'), S=last.S,

               A4_ma200w=round(float(last.get('A4_ma200w')), 4), hy_pct10y=round(float(last.hy_pct10y), 1),

def zigzag(s, th):
    """종가 지그재그. th 이상 반대 방향 움직임이 나오면 직전 극점 확정. 반환 [(날짜, 값, 'L'/'H')]"""
    v = s.values; ix = s.index; piv = []; mode = 0; hi_i = lo_i = 0
    for i in range(1, len(v)):
        if v[i] > v[hi_i]: hi_i = i
        if v[i] < v[lo_i]: lo_i = i
        if mode >= 0 and v[i] <= v[hi_i] * (1 - th) and (mode == 1 or hi_i < i):
            if mode == 0 and lo_i < hi_i: piv.append((ix[lo_i], v[lo_i], 'L'))
            piv.append((ix[hi_i], v[hi_i], 'H')); mode = -1; lo_i = i
        elif mode <= 0 and v[i] >= v[lo_i] * (1 + th) and (mode == -1 or lo_i < i):
            if mode == 0 and hi_i < lo_i: piv.append((ix[hi_i], v[hi_i], 'H'))
            piv.append((ix[lo_i], v[lo_i], 'L')); mode = 1; hi_i = i
    return piv

def downside_base(P, lab, th=-0.65):
    dd = P / P.rolling(252, min_periods=60).max() - 1
    days = P.index[dd <= th]; rows = []
    for d in days:
        i = P.index.get_loc(d); fut = P.iloc[i + 1: i + 127]
        if len(fut) < 126: continue
        rows.append(dict(date=d, dd=dd.loc[d], further=fut.min() / P.loc[d] - 1, r126=fut.iloc[-1] / P.loc[d] - 1))
    X = pd.DataFrame(rows)
    if X.empty: return X, None
    X['episode'] = (X.date.diff().dt.days.fillna(999) > 60).cumsum()
    q = lambda s: ' / '.join(f'{v*100:+.0f}%' for v in s.quantile([0.1, 0.25, 0.5, 0.75, 0.9]))
    ep = X.groupby('episode').agg(start=('date', 'min'), end=('date', 'max'), days=('date', 'size'), worst_further=('further', 'min'),
                                  med_r126=('r126', 'median'))
    summ = dict(표본=lab, 일수=len(X), 에피소드=len(ep), 추가하락_분위=q(X.further), r126_분위=q(X.r126),
                추가하락_30이상_비율=f'{(X.further <= -0.30).mean():.0%}', r126_양수_비율=f'{(X.r126 > 0).mean():.0%}')
    return ep, summ

NAIL, SPY, TD, VIX, HY, Y10, MTG, A, C, hy_pct, pos = (G[k] for k in ['NAIL','SPY','TD','VIX','HY','Y10','MTG','A','C','hy_pct','pos'])
OUT = 'nail/research/output'; c = NAIL.close
def rt(t):
    """실시간(그날까지 자료만) 지표"""
    i = pos[t]; s = SPY.close
    return dict(date=t.date().isoformat(),
        SPY_vs63low=s.loc[t] / s.iloc[max(0, i-62):i+1].min() - 1,
        SPY_dd252=s.loc[t] / s.iloc[max(0, i-251):i+1].max() - 1,
        VIX_vs63max=VIX.loc[t] / VIX.iloc[max(0, i-62):i+1].max(),
        VIX_max21_is_max63=VIX.iloc[max(0, i-20):i+1].max() >= VIX.iloc[max(0, i-62):i+1].max(),
        HY_pct=hy_pct(t), HY_chg63=(HY.loc[t]-HY.iloc[max(0,i-63)])*100,
        HY_max21_is_max63=HY.iloc[max(0, i-20):i+1].max() >= HY.iloc[max(0, i-62):i+1].max(),
        Y10_from252max_bp=(Y10.loc[t]-Y10.iloc[max(0,i-251):i+1].max())*100,
        Y10_peak_days_ago=i - int(np.argmax(Y10.iloc[max(0,i-251):i+1].values)) - max(0,i-251),
        MTG_from52wmax=MTG.loc[t]-MTG.iloc[max(0,i-251):i+1].max())
RA = pd.DataFrame([rt(pd.Timestamp(d)) for d in A.date]); RA.insert(0,'set','A 진짜 저점')
RC = pd.DataFrame([rt(pd.Timestamp(d)) for d in C.date]); RC.insert(0,'set','C 실패 저점')
RN = pd.DataFrame([rt(pd.Timestamp(d)) for d in ['2026-10-14', str(c.index[-1].date())]]); RN.insert(0,'set','현재')
RT = pd.concat([RA, RC, RN]); RT.to_csv(f'{OUT}/realtime_checks.csv', index=False, float_format='%.4f')
pd.set_option('display.width', 250); print(RT.round(3).to_string(index=False))
chk = {'HY 10년 백분위 ≥45': lambda r: r.HY_pct >= 45, 'SPY가 63일 최저 +5% 이내': lambda r: r.SPY_vs63low <= 0.05,
       'VIX 최근 21일 안에 63일 최고': lambda r: bool(r.VIX_max21_is_max63), 'HY 최근 21일 안에 63일 최고': lambda r: bool(r.HY_max21_is_max63),
       'HY 63일 +50bp 이상': lambda r: r.HY_chg63 >= 50, '10년물 252일 최고 대비 −25bp 이하': lambda r: r.Y10_from252max_bp <= -25,
       'SPY 252일 낙폭 ≤−10%': lambda r: r.SPY_dd252 <= -0.10}
rows = []
for k, f in chk.items():
    a = [f(r) for _, r in RA.iterrows()]; cc = [f(r) for _, r in RC.iterrows()]; n = [f(r) for _, r in RN.iterrows()]
    rows.append(dict(조건=k, A=f'{sum(a)}/7', A_사례=''.join('O' if x else 'x' for x in a), C=f'{sum(cc)}/{len(cc)}', 현재_1014=n[0], 마지막=n[1]))
K = pd.DataFrame(rows); K.to_csv(f'{OUT}/realtime_commonality.csv', index=False); print(K.to_string(index=False))

# 새 252일 최저 종가에서 10거래일 안의 +8%/+10% 급반등 → 그 저점이 126거래일 동안 지켜졌나
def fresh_low_rebound(th):
    r1 = c.pct_change(); dd = c / c.rolling(252, min_periods=60).max() - 1; lo252 = c.rolling(252, min_periods=60).min()
    rows = []; used = set()
    for i in range(1, len(c)):
        if r1.iloc[i] < th: continue
        j0 = max(0, i-10); w = c.iloc[j0:i]; jl = j0 + int(np.argmin(w.values)); L = c.iloc[jl]
        if L > lo252.iloc[jl] + 1e-9 or dd.iloc[jl] > -0.50 or jl in used: continue
        used.add(jl); fut = c.iloc[i+1:i+127]
        rows.append(dict(rebound=c.index[i].date().isoformat(), ret=round(r1.iloc[i]*100,1), low_date=c.index[jl].date().isoformat(), low=round(L,2),
                         dd252_at_low=round(dd.iloc[jl]*100,1), status=('지켜짐' if (fut >= L).all() else '깨짐') if len(fut) >= 126 else f'진행중({len(fut)}일)',
                         min_after=round((fut.min()/L-1)*100,1) if len(fut) else None, days_to_break=(int(np.argmax(fut.values < L))+1) if len(fut) and (fut < L).any() else None))
    return pd.DataFrame(rows)
for th in (0.08, 0.10):
    F = fresh_low_rebound(th); F.to_csv(f'{OUT}/fresh_low_rebound_{int(th*100)}.csv', index=False)
    done = F[~F.status.str.startswith('진행')]
    print(f'\n=== 새 1년 최저 종가 후 10거래일 안 +{int(th*100)}% 급반등: 완료 {len(done)}건 중 지켜짐 {int((done.status=="지켜짐").sum())} ===')
    print(F.to_string(index=False))

# 252일 낙폭 −65% 첫 도달 에피소드 → 이후 252거래일 최저까지
dd = c / c.rolling(252, min_periods=60).max() - 1
first = []; last_i = -999
for i in range(len(c)):
    if dd.iloc[i] <= -0.65:
        if i - last_i > 63: first.append(i)
        last_i = i
rows = []
for i in first:
    fut = c.iloc[i+1:i+253]
    rows.append(dict(first_day=c.index[i].date().isoformat(), close=round(c.iloc[i],2), dd=round(dd.iloc[i]*100,1),
                     min_252=round((fut.min()/c.iloc[i]-1)*100,1) if len(fut) else None, min_date=fut.idxmin().date().isoformat() if len(fut) else None,
                     days_to_min=int(np.argmin(fut.values))+1 if len(fut) else None, fwd_days=len(fut)))
E = pd.DataFrame(rows); E.to_csv(f'{OUT}/first_cross_65.csv', index=False); print('\n=== 252일 낙폭 −65% 첫 도달(63거래일 공백으로 구분) ===\n', E.to_string(index=False))

print('\n최근 금리:', Y10.loc['2026-08-01':].dropna().resample('W-FRI').last().round(2).to_dict())
print('10년물 252일 최고:', Y10.loc['2025-10-15':].max(), Y10.loc['2025-10-15':].idxmax().date())
