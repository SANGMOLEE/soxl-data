## 3. 매수 규칙 v1 (결과를 보기 전의 제안)

| 차수 | 조건 (신호일 종가 확인 → 다음 거래일 시가 체결) | 예산 |
|---|---|---:|

## 4. 평가 방법 (고정)

**단기 저점의 정의.** 날짜 t가 단기 저점이려면 ① 종가가 t−10 ~ t+10 거래일 중 최저 ② t+1 ~ t+20 거래일 최고 종가가 t 종가의 1.06배 이상 ③ t 종가가 t−30 ~ t 최고 종가의 0.93배 이하(하락 뒤의 저점일 것). 2011-04~2026-09 전 기간에서 추출한다.

**지표.**
1. 단기 저점마다 저점일 표시점수, 저점 ±3일 최저 표시점수, 블록별 구성.
2. 기준선 X ∈ {70, 65, 60, 55, 50, 45, 40}마다 **에피소드**(X 이하로 처음 내려간 날; 직전 20거래일 안에 같은 신호가 없을 때) 기준으로: 신호 수, 신호일 다음 시가 체결 뒤 5/10/20거래일 수익률(중앙·평균·승률), 20거래일 안 최저 종가까지의 추가 하락(중앙·최악), 신호 ±5거래일 안에 단기 저점이 있는 비율, 단기 저점까지 거래일 수 중앙.
3. 같은 지표를 전체 거래일(기준선 없음)과 비교한다.
4. 블록 기여: P만, P+S, P+S+G, 전체(V 포함, 2015년 이후만)로 2번을 반복한다.
5. 안정성: 2011~2018 / 2019~2026 두 구간으로 나눠 2번을 따로 적는다.
6. 표본이 15개 미만인 표는 사례를 하나씩 나열한다. 신뢰구간은 클로퍼-피어슨 95%.

**결론 형식.** 10/16 종가의 표시점수와 구성, 과거 단기 저점의 점수 분포 대비 위치, 다음 거래일에 기준선을 넘기 위한 조건(블록별 경계값), 그리고 "저점 근처일 가능성이 높다 / 판단 불가 / 아직 아니다" 중 하나.

# ─────────────────────────────── 2b. 네이버 금융 (KRX가 해외 IP를 막을 때의 대안)
import time, io, re
try:
    import requests
    NH = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 Chrome/124.0 Safari/537.36",
          "Referer": "https://finance.naver.com/"}
    os.makedirs(os.path.join(OUT, "naver"), exist_ok=True)
    SESSION = requests.Session(); SESSION.headers.update(NH)

    def get(url, **kw):
        r = SESSION.get(url, timeout=30, **kw); r.raise_for_status(); return r

    def read_tables(html, **kw):
        try:
            return pd.read_html(io.StringIO(html), flavor="lxml", **kw)
        except Exception:
            return pd.read_html(io.StringIO(html), flavor="bs4", **kw)

    # 2b-1. 코스피 투자자별 일별 매매동향 (억원) — 개인/외국인/기관계 ...
    START = pd.Timestamp("2015-01-01")
    def naver_investor_trend(sosok, label):
        rows = []
        bizdate = datetime.date.today().strftime("%Y%m%d")
        for page in range(1, 400):
            url = f"https://finance.naver.com/sise/investorDealTrendDay.naver?bizdate={bizdate}&sosok={sosok}&page={page}"
            r = get(url); r.encoding = "euc-kr"
            tabs = read_tables(r.text, match="날짜")
            if not tabs: break
            t = tabs[0]
            if isinstance(t.columns, pd.MultiIndex):
                t.columns = ["_".join([str(c) for c in col if not str(c).startswith("Unnamed")]).strip("_") for col in t.columns]
            t = t.dropna(how="all")
            t = t[t.iloc[:, 0].astype(str).str.match(r"\d{2}\.\d{2}\.\d{2}")]
            if len(t) == 0: break
            rows.append(t)
            last = pd.to_datetime(t.iloc[-1, 0], format="%y.%m.%d")
            if last < START: break
            time.sleep(0.15)
        if not rows: fail(f"naver investor {label}: 빈 응답"); return
        df = pd.concat(rows, ignore_index=True)
        df = df.rename(columns={df.columns[0]: "date"})
        df["date"] = pd.to_datetime(df["date"], format="%y.%m.%d").dt.strftime("%Y-%m-%d")
        df = df.drop_duplicates("date").sort_values("date")
        df = df[pd.to_datetime(df["date"]) >= START]

def bands_le(x, cuts, pts):
    x = np.asarray(x, float); out = np.zeros(len(x))
    for c, p in zip(cuts, pts): out = np.where((x <= c) & (p > out), p, out)
    return np.where(np.isnan(x), 0.0, out)

def bands_ge(x, cuts, pts):
    x = np.asarray(x, float); out = np.zeros(len(x))
    for c, p in zip(cuts, pts): out = np.where((x >= c) & (p > out), p, out)
    return np.where(np.isnan(x), 0.0, out)

def wilder_rsi(x, n=14):
    d = x.diff().values; up = np.clip(d, 0, None); dn = np.clip(-d, 0, None)
    out = np.full(len(x), np.nan)
    if len(x) <= n: return pd.Series(out, index=x.index)
    au = np.nanmean(up[1:n + 1]); ad = np.nanmean(dn[1:n + 1])
    out[n] = 100 - 100 / (1 + au / ad) if ad > 0 else 100.0
    for i in range(n + 1, len(x)):
        au = (au * (n - 1) + up[i]) / n; ad = (ad * (n - 1) + dn[i]) / n
        out[i] = 100 - 100 / (1 + au / ad) if ad > 0 else 100.0
    return pd.Series(out, index=x.index)

def asof_strict(series, tdays):
    """날짜 < t 인 마지막 값 (한국 아침에 아는 미국 값)."""
    s = series.dropna(); idx = s.index.searchsorted(tdays, side='left') - 1

def rolling_pct(x, n=250, minp=120):
    """x의 직전 n개(당일 제외) 대비 백분위(0~100) = 과거값 중 x 이하 비율."""
    v = x.values; out = np.full(len(v), np.nan)
    for i in range(len(v)):
        past = v[max(0, i - n):i]; past = past[~np.isnan(past)]
        if len(past) >= minp and not np.isnan(v[i]): out[i] = 100.0 * np.mean(past <= v[i])
    return pd.Series(out, index=x.index)

def simulate(df, rule=BUY_RULE):
    fills = []; tranche = 0; last_px = np.nan; last_i = -99; armed = True
    disp = df["display"].values; close = df["close"].values; opn = df["open"].values; dd = df["p1_dd20"].values
    for i in range(len(df) - 1):
        if tranche > 0 and dd[i] > rule["rearm_dd"]:
            tranche = 0; last_px = np.nan; last_i = -99
        if np.isnan(disp[i]): continue
        if tranche == 0 and disp[i] <= rule["t1"]:
            fills.append((df.index[i], df.index[i + 1], 1, rule["w"][0], opn[i + 1], disp[i])); tranche = 1; last_px = opn[i + 1]; last_i = i
        elif tranche in (1, 2):
            thr = rule["t2"] if tranche == 1 else rule["t3"]
            if disp[i] <= thr and close[i] <= last_px * rule["gap"] and i - last_i >= rule["min_days"]:
                fills.append((df.index[i], df.index[i + 1], tranche + 1, rule["w"][tranche], opn[i + 1], disp[i])); tranche += 1; last_px = opn[i + 1]; last_i = i
    return pd.DataFrame(fills, columns=["signal", "fill", "tranche", "weight", "fill_px", "display_at_signal"])

from scipy.stats import beta

def clopper(k, n, a=0.05):
    lo = beta.ppf(a / 2, k, n - k + 1) if k > 0 else 0.0
    hi = beta.ppf(1 - a / 2, k + 1, n - k) if k < n else 1.0
    return lo, hi

def short_bottoms(close, w=10, fwd=20, rebound=0.06, decline=0.93, back=30):
    """§4 정의: ±w 최저 종가, 이후 fwd일 최고 ≥ 1+rebound, 직전 back일 최고 대비 ≤ decline."""
    c = close.values; n = len(c); out = []
    for t in range(back, n - fwd):
        if c[t] != c[t - w:t + w + 1].min(): continue
        if c[t + 1:t + fwd + 1].max() < c[t] * (1 + rebound): continue
        if c[t] > c[t - back:t + 1].max() * decline: continue
        out.append(t)
    return out

def episodes(disp, thr, gap=20):
    """thr 이하로 처음 내려간 날(직전 gap일 안에 같은 신호 없음)."""
    idx = []; last = -10**9
    v = disp.values
    for i in range(len(v)):
        if not np.isnan(v[i]) and v[i] <= thr and i - last > gap:
            idx.append(i); last = i
    return idx

def fwd_stats(df, idxs, bottoms, label):
    """신호일 i → 다음날 시가 체결. 5/10/20일 수익률, 20일 최저 종가 추가 하락, ±5일 안 단기 저점, 다음 저점까지 일수."""
    c = df["close"].values; o = df["open"].values; n = len(c); rows = []
    bset = np.array(bottoms)
    for i in idxs:
        if i + 1 >= n: continue
        px = o[i + 1]
        r = {"signal": df.index[i].date(), "display": df["display"].values[i], "fill_px": px}
        for h in (5, 10, 20):
            r[f"ret{h}"] = c[i + 1 + h] / px - 1 if i + 1 + h < n else np.nan
        r["mae20"] = c[i + 1:i + 21].min() / px - 1 if i + 21 <= n else np.nan
        r["mfe20"] = c[i + 1:i + 21].max() / px - 1 if i + 21 <= n else np.nan
        near = bset[(bset >= i - 5) & (bset <= i + 5)]
        r["bottom_within5"] = len(near) > 0
        nxt = bset[bset >= i]
        r["days_to_bottom"] = int(nxt[0] - i) if len(nxt) else np.nan
        r["bottom_vs_fill"] = c[nxt[0]] / px - 1 if len(nxt) else np.nan
        rows.append(r)
    out = pd.DataFrame(rows); out["set"] = label
    return out

def summarize(tab, name):
    if len(tab) == 0: return {"set": name, "n": 0}
    done = tab.dropna(subset=["ret20"])
    k = int(tab["bottom_within5"].sum()); n = len(tab); lo, hi = clopper(k, n)
    return {"set": name, "n": n,
            "ret5_med": done["ret5"].median(), "ret10_med": done["ret10"].median(), "ret20_med": done["ret20"].median(),
            "ret20_mean": done["ret20"].mean(), "win20": (done["ret20"] > 0).mean(),
            "mae20_med": done["mae20"].median(), "mae20_worst": done["mae20"].min(),
            "bottom5_rate": k / n, "bottom5_ci": f"{lo:.0%}~{hi:.0%}", "days_to_bottom_med": tab["days_to_bottom"].median(),
            "bottom_vs_fill_med": tab["bottom_vs_fill"].median()}

# ───────────────────────────── 급락형 저점의 점수 경로 (사후 분석용 보조)
def crash_paths(session, repo, min_drop=-0.17, out_name="crash_bottom_paths.csv"):
    df, meta = compute(session, repo); df = df[df.index >= "2011-06-01"]
    bottoms = short_bottoms(df["close"]); rows = []
    for t in bottoms:
        drop = df["close"].iloc[t] / df["close"].iloc[t - 30:t + 1].max() - 1
        if drop > min_drop: continue
        path = df["display"].iloc[t - 7:t + 6].round(0).astype(int).tolist()
        r = {"bottom": df.index[t].date(), "close": df["close"].iloc[t], "drop30": round(drop, 3),
             "rebound20": round(df["close"].iloc[t + 1:t + 21].max() / df["close"].iloc[t] - 1, 3),
             "display_at_bottom": int(df["display"].iloc[t]), "path_m7_p5": path}
        for thr in (60, 50, 45, 40):
            win = df["display"].iloc[t - 15:t + 1]; hit = win[win <= thr]
            r[f"first_le{thr}_rel"] = int(df.index.get_loc(hit.index[0]) - t) if len(hit) else None
            r[f"close_at_first_le{thr}_vs_bottom"] = round(df["close"].loc[hit.index[0]] / df["close"].iloc[t] - 1, 3) if len(hit) else None
        rows.append(r)
    out = pd.DataFrame(rows); out.to_csv(os.path.join(session, "research", "output", out_name), index=False)
    return out
