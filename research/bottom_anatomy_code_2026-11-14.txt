def rsi(s, n=14):
    d = s.diff(); u = d.clip(lower=0).ewm(alpha=1 / n, adjust=False).mean(); dn = (-d.clip(upper=0)).ewm(alpha=1 / n, adjust=False).mean(); return 100 - 100 / (1 + u / dn)

def feats(b):
    F = b[["open", "high", "low", "close", "volume"]].copy(); c = F.close
    F["ret"] = c.pct_change(); F["sd60"] = F.ret.rolling(60).std().shift(1); F["z"] = F.ret / F.sd60
    F["vr"] = F.volume / F.volume.rolling(30).median().shift(1); F["rsi"] = rsi(c)
    F["ath"] = F.high.cummax(); F["dd"] = c / F.ath - 1; F["zone"] = F.dd <= -0.45
    F["cap"] = (F.z <= -2.5) & (F.vr >= 1.5); F["fin"] = (F.z <= -5) & F.zone
    F["hi30"] = F.high.rolling(30).max(); F["fall30"] = F.low / F.hi30 - 1
    return F

def state_machine(F, window=180):
    """그날까지의 정보로 본 국면. 0 피날레 없음 / 1 투매 진행(마지막 투매일 7일 미만) / 2 재시험 대기 / 3 조용한 재이탈 신호."""
    idx = F.index; fin_days = list(F.index[F.fin]); cap_days = list(F.index[F.cap])
    out = pd.DataFrame(index=idx, columns=["phase", "L", "Ldate", "Rmin", "since_cap", "since_fin", "cl_start"]); out["phase"] = 0
    fi = 0; ci = 0; last_fin = None; cl_start = None; last_cap = None
    lows = F.low.values; rs = F.rsi.values; pos = {d: i for i, d in enumerate(idx)}
    for i, d in enumerate(idx):
        while fi < len(fin_days) and fin_days[fi] <= d:
            f = fin_days[fi]
            if last_fin is None or (f - last_fin).days > 30: cl_start = f
            last_fin = f; fi += 1
        while ci < len(cap_days) and cap_days[ci] <= d: last_cap = cap_days[ci]; ci += 1
        if last_fin is None or (d - last_fin).days > window: continue
        s = pos[cl_start]
        if i <= s: out.iat[i, 0] = 1; out.iat[i, 5] = 0; out.iat[i, 6] = cl_start; continue
        seg = lows[s:i]; k = int(np.argmin(seg)); L = seg[k]; Rmin = np.nanmin(rs[s:i])
        since = (d - last_cap).days if last_cap is not None else 999
        quiet = bool(F.zone.iat[i]) and since >= 7 and lows[i] <= L * 1.03 and F.vr.iat[i] <= 1.3 and rs[i] >= Rmin + 5
        out.iloc[i] = [3 if quiet else (1 if since < 7 else 2), L, idx[s + k], Rmin, since, (d - last_fin).days, cl_start]
    return out

# ───────────────────────── 3. 5σ 음봉 사건 전수 ─────────────────────────
def sigma_events(F, label, mv):
    c, l = F.close, F.low
    P(f"\n== 3. 약세 구간의 5σ 음봉 전수 조사 ({label}). 추가 하락 = 그날 종가 대비 이후 1년 최저 저가")
    ev = F.index[F.fin]; res = []
    for d in ev:
        fut = l[d + TD(days=1):d + TD(days=365)]
        if not len(fut): continue
        more = min(fut.min() / c[d] - 1, 0); dl = (fut.idxmin() - d).days if fut.min() < c[d] else 0
        f = lambda n: c.asof(d + TD(days=n)) / c[d] - 1 if d + TD(days=n) <= c.index[-1] else np.nan
        res.append(more)
        P(f"   {d.date()} ${c[d]:>9,.0f} | 종가 {F.ret[d]:+6.1%} = {F.z[d]:5.1f}σ | 거래량 {F.vr[d]:4.1f}배 | 고점대비 {F.dd[d]:+.0%} | 30일 고점대비(저가) {F.fall30[d]:+.0%} | RSI {F.rsi[d]:.0f} | MVRV {mv.asof(d):.2f} | 추가 하락 {more:+4.0%} ({dl}일 뒤) | 30일 {f(30):+4.0%} 90일 {f(90):+4.0%} 1년 {f(365):+5.0%}")
    base = []
    for d in c[F.zone].index:
        fut = l[d + TD(days=1):d + TD(days=365)]
        if len(fut) >= 300: base.append(min(fut.min() / c[d] - 1, 0))
    base = np.array(base); res = np.array(res)
    P(f"   → 5σ 음봉 종가 매수 n={len(res)}: 추가 하락 중앙값 {np.median(res):+.0%}, -10% 이내 {np.mean(res >= -0.10):.0%}, 최악 {res.min():+.0%}")
    P(f"   → 기준선(약세 구간 아무 날 n={len(base)}): 추가 하락 중앙값 {np.median(base):+.0%}, -10% 이내 {np.mean(base >= -0.10):.0%}")
    capm = []; last = None
    for d in F.index[F.cap & F.zone]:
        if last is not None and (d - last).days < 20: continue
        last = d; fut = l[d + TD(days=1):d + TD(days=365)]
        if len(fut): capm.append(min(fut.min() / c[d] - 1, 0))
    capm = np.array(capm)
    P(f"   → 비교: 일반 투매일(2.5σ·거래량 1.5배, 20일 간격) 종가 매수 n={len(capm)}: 추가 하락 중앙값 {np.median(capm):+.0%}, -10% 이내 {np.mean(capm >= -0.10):.0%}, 최악 {capm.min():+.0%}")

def act_table(F):

    fng = pd.read_csv(os.path.join(EX, "FNG.csv"), parse_dates=["date"]).set_index("date").value.sort_index(); fng = fng[~fng.index.duplicated()]
    fu = pd.read_csv(os.path.join(EX, "FUNDING_BINANCE.csv"), parse_dates=["date"]).set_index("date").rate.sort_index().resample("D").sum() * 1e4
    dv = pd.read_csv(os.path.join(EX, "DVOL.csv"), parse_dates=["date"]).set_index("date").close
    mv = pd.read_csv(os.path.join(DC, "BTC_ONCHAIN.csv"), parse_dates=["date"]).set_index("date").mvrv

# ───────────────────────── 5. 주요 스윙 저점 24곳 통계 ─────────────────────────
def swing_stats(F):
    c, h, l = F.close, F.high, F.low; rows = []
    for d in l.index[90:-10]:
        if l[d] > l[d - TD(days=45):d + TD(days=45)].min(): continue
        if l[d] / h[d - TD(days=90):d].max() - 1 > -0.25: continue
        pre = F.loc[d - TD(days=30):d]; wd = pre.z.idxmin()
        rows.append(dict(date=d, gap=(d - wd).days, z=pre.z.min(), green=F.ret[d] > 0, clv=(c[d] - l[d]) / (h[d] - l[d]), vr=F.vr[d], rsi_div=F.rsi[d] - pre.rsi.min(), under=l[d] / l[wd] - 1))
    S = pd.DataFrame(rows)
    P(f"\n== 5. 주요 저점 {len(S)}곳 통계 (앞뒤 45일 최저가이면서 90일 고점 대비 -25% 이하, 2014-12~)")
    P(f"   직전 30일 최대 음봉(σ 기준)과 저점의 간격: 같은 날 {int((S.gap == 0).sum())}곳, 1일 뒤 {int((S.gap == 1).sum())}곳, 2~8일 뒤 {int(((S.gap >= 2) & (S.gap <= 8)).sum())}곳, 9일 이상 뒤 {int((S.gap >= 9).sum())}곳")
    P(f"   저점 당일이 양봉으로 끝난 곳 {int(S.green.sum())}/{len(S)}, 종가가 일중 범위 위쪽 절반에서 끝난 곳 {int((S.clv >= 0.5).sum())}/{len(S)}")
    P(f"   저점 당일 거래량이 30일 중앙값의 1.5배 미만인 곳 {int((S.vr < 1.5).sum())}/{len(S)}, 저점일 RSI가 직전 30일 최저 RSI보다 5 이상 높은 곳(상승 다이버전스) {int((S.rsi_div >= 5).sum())}/{len(S)}")
    P(f"   최대 음봉의 저가 대비 최종 저점 깊이: 중앙값 {S.under.median():+.1%}, 사분위 {S.under.quantile(.25):+.1%} ~ {S.under.quantile(.75):+.1%}")

# ───────────────────────── 6. 국면 추적기 검증 ─────────────────────────
def phase_backtest(F, SM, label):
    c, l = F.close, F.low
    P(f"\n== 6. 국면 추적기 검증 ({label}): 피날레 음봉 묶음별로 '종가 매수' 와 '조용한 재이탈 첫 신호' 를 최종 저점과 비교")
    starts = sorted(set(SM.cl_start.dropna()))
    for s in starts:
        seg = SM[SM.cl_start == s]; end = seg.index[-1]; fins = F.index[(F.fin) & (F.index >= s) & (F.index <= end)]
        fl = l[s:end + TD(days=1)].min(); fd = l[s:end + TD(days=1)].idxmin()
        fut = l[fd:fd + TD(days=365)].min()

def now_state(F, SM, mv):

    recent_cap = [str(k.date()) for k in F.index[F.cap][-3:]]
    P(f"\n== 8. 현재 상태 {d.date()} (마지막 봉은 진행 중일 수 있음)")

    P(f"   최근 투매일(2.5σ·거래량 1.5배): {recent_cap}")
    P(f"   오늘 기준 직전 60일 변동성 {sd:.2%}/일 → 마지막 봉(진행 중)이 피날레 음봉(5σ)으로 끝나려면 종가 ${fin_px:,.0f} 이하({-5 * sd:+.1%}), 투매일(2.5σ)은 ${cap_px:,.0f} 이하. 약세 구간 경계(고점 -45%) ${zone_px:,.0f}")
    st = dict(date=str(d.date()), close=round(float(x.close)), ret=round(float(x.ret), 4), z60=round(float(x.z), 2), vol_ratio=round(float(x.vr), 2), rsi_d=round(float(x.rsi), 1), dd_ath=round(float(x.dd), 4),
              zone=bool(x.zone), phase=int(s.phase), phase_name=names[int(s.phase)], finale_today=bool(x.fin), cap_today=bool(x.cap), sd60=round(sd, 4),
              next_finale_close=round(fin_px), next_cap_close=round(cap_px), zone_price=round(zone_px),
              L=None if not int(s.phase) or pd.isna(s.L) else round(float(s.L)), L_date=None if not int(s.phase) or pd.isna(s.L) else str(pd.Timestamp(s.Ldate).date()),
              retest_below=None if not int(s.phase) or pd.isna(s.L) else round(float(s.L) * 1.03), recent_cap_days=recent_cap, mvrv=round(float(mv.asof(d)), 3))
    json.dump(st, open(os.path.join(HERE, "bottom_phase_state.json"), "w"), ensure_ascii=False, indent=1)

def get(url, params=None, tries=4, timeout=60):
    last = None
    for k in range(tries):
        try:
            r = S.get(url, params=params, timeout=timeout)
            if r.status_code == 200: return r
            last = f"HTTP {r.status_code}"
            if r.status_code in (400, 403, 404, 451): break
        except Exception as e: last = repr(e)
        time.sleep(2 + 3 * k)
    raise RuntimeError(f"{url} -> {last}")
NOW = int(time.time())

def bitstamp(step, name, start="2011-08-01", gz=False):
    ts = int(pd.Timestamp(start).timestamp()); rows = []
    while ts < NOW:
        j = get("https://www.bitstamp.net/api/v2/ohlc/btcusd/", dict(step=step, limit=1000, start=ts)).json()["data"]["ohlc"]
        if not j: ts += step * 1000; continue
        rows += j; last = int(j[-1]["timestamp"])
        ts = last + step if last >= ts else ts + step * 1000
        time.sleep(0.25)
    d = pd.DataFrame(rows).drop_duplicates("timestamp"); d["date"] = pd.to_datetime(d["timestamp"].astype(int), unit="s")
    d = d[["date", "open", "high", "low", "close", "volume"]].sort_values("date")
    d = d[pd.to_numeric(d["volume"]) > 0]
    p = os.path.join(OUT, name + (".csv.gz" if gz else ".csv")); d.to_csv(p, index=False)
    log(f"[OK] {name}: {len(d)} rows {d.date.min()} ~ {d.date.max()}")

    head = None if raw[:1].isdigit() else 0
